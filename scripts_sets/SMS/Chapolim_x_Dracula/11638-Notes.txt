0x000191: [8-bit] Game State
0x10 = Player
0x00 = Demo mode

0x0001ab: [8-bit] P1 Input Controller
Bit0 = Up
Bit1 = Down
Bit2 = Left
Bit3 = Right
Bit4 = Button 2
Bit5 = Button 1

0x0001bb: [8-bit] Game State
0x05 = In Game
0x00 = Menu game

0x0001df: [8-bit] Stage ID
0x01 = Stage 1
0x02 = Stage 2
0x03 = Stage 3
0x04 = Stage 4
0x05 = Stage 5
0x06 = Stage 6
0x07 = Victory Screen / Ending

0x0001e0: [8-bit] Lives Count
0x00 = 0 Lives (Last chance)
0x01 = 1 Life
0x02 = 2 Lives
0x03 = 3 Lives (Extra)
0xff = Game Over Screen

0x0001e8: [24-bit] Player Score
[Note: Stored in BCD Little Endian. Multiply by 10 for on-screen value. Example: 0x005145 in RAM = 51450 on screen]

Bit2 = 1st Treasure Collected

Bit4 = 12th Treasure Collected
Bit5 = 11th Treasure Collected

0x00021b: [8-bit] Chapolim HP
